// ------------------------------------------------------------ // LOCAL STATE // ------------------------------------------------------------ let reviewFlights = []; // ------------------------------------------------------------ // REDIRECT // ------------------------------------------------------------ function redirectToLogin() { window.location.replace( "/management/" ); } // ------------------------------------------------------------ // DISPLAY ROLE // ------------------------------------------------------------ function formatRole(role) { if (!role) { return "MANAGEMENT"; } return role .replace(/_/g, " ") .toUpperCase(); } // ------------------------------------------------------------ // ERROR HANDLING // ------------------------------------------------------------ function showError(message) { if (!reviewError) { return; } reviewError.textContent = message; reviewError.classList.remove( "hidden" ); } function hideError() { if (!reviewError) { return; } reviewError.textContent = ""; reviewError.classList.add( "hidden" ); } // ------------------------------------------------------------ // MANAGEMENT AUTHORISATION // ------------------------------------------------------------ async function getManagementUser(userId) { const { data, error } = await supabaseClient .from("management_users") .select( "display_name, role, active" ) .eq( "user_id", userId ) .eq( "active", true ) .maybeSingle(); if (error) { console.error( "Management verification failed:", error ); return null; } return data; } // ------------------------------------------------------------ // FORMAT FLIGHT TIME // ------------------------------------------------------------ function formatMinutes(minutes) { const value = Number(minutes) || 0; const hours = Math.floor( value / 60 ); const remaining = value % 60; return `${hours}h ${remaining}m`; } // ------------------------------------------------------------ // FORMAT LANDING RATE // ------------------------------------------------------------ function formatLandingRate(value) { if ( value === null || value === undefined ) { return "—"; } return `${value} fpm`; } // ------------------------------------------------------------ // LANDING CLASS // ------------------------------------------------------------ function getLandingClass(value) { if ( value === null || value === undefined ) { return ""; } const landing = Math.abs( Number(value) ); if (landing >= 500) { return "hard"; } if (landing <= 250) { return "good"; } return ""; } // ------------------------------------------------------------ // ESCAPE HTML // ------------------------------------------------------------ function escapeHtml(value) { if ( value === null || value === undefined ) { return ""; } return String(value) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); } // ------------------------------------------------------------ // LOAD ALL PIREPS // ------------------------------------------------------------ async function loadPireps() { console.log( "Loading British Midland PIREPs..." ); const { data, error } = await supabaseClient .from("pireps") .select(` id, pilot_id, flight_number, departure, arrival, aircraft, registration, block_minutes, landing_rate_fpm, status, submitted_at, flight_score, flight_grade, performance_rating, landing_assessment, score_deduction, requires_review `) .order( "submitted_at", { ascending: false } ); if (error) { console.error( "Unable to load PIREPs:", error ); throw error; } console.log( "PIREPs loaded:", data?.length || 0 ); return data || []; } // ------------------------------------------------------------ // LOAD PILOTS // ------------------------------------------------------------ async function loadPilots() { const { data, error } = await supabaseClient .from("pilots") .select(` id, pilot_id, nickname `); if (error) { console.error( "Unable to load pilot records:", error ); throw error; } return data || []; } // ------------------------------------------------------------ // BUILD PILOT LOOKUP // ------------------------------------------------------------ function buildPilotLookup(pilots) { const lookup = new Map(); pilots.forEach( pilot => { lookup.set( pilot.id, pilot ); } ); return lookup; } // ------------------------------------------------------------ // STATISTICS // ------------------------------------------------------------ function updateStatistics(allPireps) { const total = allPireps.length; const reviews = allPireps.filter( pirep => pirep.requires_review === true ); const scored = allPireps.filter( pirep => pirep.flight_score !== null && pirep.flight_score !== undefined ); let average = null; if (scored.length > 0) { const scoreTotal = scored.reduce( (sum, pirep) => sum + Number( pirep.flight_score ), 0 ); average = Math.round( scoreTotal / scored.length ); } const rate = total > 0 ? ( reviews.length / total ) * 100 : 0; requiresReviewCount.textContent = reviews.length; totalPirepCount.textContent = total; averageScore.textContent = average !== null ? `${average}%` : "—"; reviewRate.textContent = `${rate.toFixed(1)}%`; } // ------------------------------------------------------------ // CREATE REVIEW RECORDS // ------------------------------------------------------------ function createReviewRecords( allPireps, pilotLookup ) { return allPireps .filter( pirep => pirep.requires_review === true ) .map( pirep => { const pilot = pilotLookup.get( pirep.pilot_id ); return { ...pirep, pilotNumber: pilot?.pilot_id || "UNKNOWN", pilotName: pilot?.nickname || "Unknown Pilot" }; } ); } // ------------------------------------------------------------ // RENDER QUEUE // ------------------------------------------------------------ function renderReviewQueue(flights) { if (!reviewQueue) { return; } if (flights.length === 0) { reviewQueue.innerHTML = `
Review queue clear No flights currently require management attention.
`; return; } reviewQueue.innerHTML = flights .map( flight => { const landingClass = getLandingClass( flight.landing_rate_fpm ); const aircraft = flight.registration ? `${flight.aircraft || "—"} · ${flight.registration}` : ( flight.aircraft || "—" ); const score = flight.flight_score !== null && flight.flight_score !== undefined ? `${flight.flight_score}%` : "—"; return `
${escapeHtml(flight.pilotName)} ${escapeHtml(flight.pilotNumber)}
${escapeHtml(flight.flight_number)}
${escapeHtml(flight.departure)} → ${escapeHtml(flight.arrival)}
${escapeHtml(aircraft)}
${escapeHtml( formatLandingRate( flight.landing_rate_fpm ) )}
${escapeHtml(score)}
REQUIRES REVIEW
`; } ) .join(""); attachReviewRowEvents(); } // ------------------------------------------------------------ // OPEN FLIGHT REPORT // ------------------------------------------------------------ function openFlightReport(pirepId) { if (!pirepId) { return; } /* * Individual flight report page. * * The report receives the PIREP UUID through * the URL query parameter. */ window.location.href = `/management//management/report/?id=${encodeURIComponent( pirepId )}`; } // ------------------------------------------------------------ // ROW EVENTS // ------------------------------------------------------------ function attachReviewRowEvents() { const rows = document.querySelectorAll( ".review-row" ); rows.forEach( row => { row.addEventListener( "click", () => { openFlightReport( row.dataset.pirepId ); } ); row.addEventListener( "keydown", event => { if ( event.key === "Enter" || event.key === " " ) { event.preventDefault(); openFlightReport( row.dataset.pirepId ); } } ); } ); } // ------------------------------------------------------------ // FILTER QUEUE // ------------------------------------------------------------ function filterReviewQueue() { const search = reviewSearch.value .trim() .toLowerCase(); if (!search) { renderReviewQueue( reviewFlights ); return; } const filtered = reviewFlights.filter( flight => { const searchable = [ flight.pilotNumber, flight.pilotName, flight.flight_number, flight.departure, flight.arrival, flight.aircraft, flight.registration, flight.flight_grade, flight.performance_rating, flight.landing_assessment ] .filter(Boolean) .join(" ") .toLowerCase(); return searchable.includes( search ); } ); renderReviewQueue( filtered ); } // ------------------------------------------------------------ // LOAD REVIEW CENTRE // ------------------------------------------------------------ async function loadReviewCentre() { hideError(); try { const [ allPireps, pilots ] = await Promise.all([ loadPireps(), loadPilots() ]); const pilotLookup = buildPilotLookup( pilots ); updateStatistics( allPireps ); reviewFlights = createReviewRecords( allPireps, pilotLookup ); renderReviewQueue( reviewFlights ); console.log( "Review queue loaded:", reviewFlights.length ); } catch (error) { console.error( "Review Centre loading failed:", error ); reviewQueue.innerHTML = `
Unable to load review queue Operations Control could not retrieve the ACARS review data.
`; showError( "Unable to retrieve Review Centre data from ACARS." ); } } // ------------------------------------------------------------ // INITIALISE PAGE // ------------------------------------------------------------ async function initialiseReviews() { try { // ---------------------------------------------------- // CHECK SUPABASE SESSION // ---------------------------------------------------- const { data: sessionData, error: sessionError } = await supabaseClient .auth .getSession(); if ( sessionError || !sessionData.session ) { redirectToLogin(); return; } const user = sessionData.session.user; // ---------------------------------------------------- // VERIFY MANAGEMENT ACCESS // ---------------------------------------------------- const manager = await getManagementUser( user.id ); if (!manager) { await supabaseClient .auth .signOut(); redirectToLogin(); return; } console.log( "Review Centre access granted:", { displayName: manager.display_name, role: manager.role } ); // ---------------------------------------------------- // MANAGEMENT DISPLAY // ---------------------------------------------------- managementName.textContent = manager.display_name; managementRole.textContent = formatRole( manager.role ); // ---------------------------------------------------- // REVEAL APPLICATION // ---------------------------------------------------- authGate.classList.add( "hidden" ); dashboardApp.classList.remove( "hidden" ); // ---------------------------------------------------- // LOAD REVIEW DATA // ---------------------------------------------------- await loadReviewCentre(); } catch (error) { console.error( "Review Centre initialisation failed:", error ); showError( "Operations Control could not initialise the Review Centre." ); } } // ------------------------------------------------------------ // SIGN OUT // ------------------------------------------------------------ async function signOut() { if (signOutButton) { signOutButton.disabled = true; } try { await supabaseClient .auth .signOut(); } finally { redirectToLogin(); } } // ------------------------------------------------------------ // EVENT LISTENERS // ------------------------------------------------------------ if (reviewSearch) { reviewSearch.addEventListener( "input", filterReviewQueue ); } if (signOutButton) { signOutButton.addEventListener( "click", signOut ); } supabaseClient.auth.onAuthStateChange( event => { if (event === "SIGNED_OUT") { redirectToLogin(); } } ); // ------------------------------------------------------------ // START REVIEW CENTRE // ------------------------------------------------------------ initialiseReviews();
